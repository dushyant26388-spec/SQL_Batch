Q26. Display the salary of the first employee hired in each department.

SELECT name, dept, hire_year, salary,
       FIRST_VALUE(salary) OVER(
           PARTITION BY dept ORDER BY hire_year, name
       ) AS first_hired_salary
FROM employees;

Q27. Display the salary of the last employee hired in each department.

SELECT name, dept, hire_year, salary,
       LAST_VALUE(salary) OVER(
           PARTITION BY dept
           ORDER BY hire_year, name
           ROWS BETWEEN UNBOUNDED PRECEDING AND UNBOUNDED FOLLOWING
       ) AS last_hired_salary
FROM employees;

Q28. Display the salary of the employee hired immediately before each employee in the same department.

SELECT name, dept, hire_year, salary,
       LAG(salary, 1) OVER(
           PARTITION BY dept ORDER BY hire_year, name
       ) AS previous_employee_salary
FROM employees;

Q29. Display the salary of the employee hired two positions before each employee.

SELECT name, dept, hire_year, salary,
       LAG(salary, 2) OVER(
           PARTITION BY dept ORDER BY hire_year, name
       ) AS salary_two_positions_before
FROM employees;

Q30. Display the salary of the employee hired two positions after each employee.

SELECT name, dept, hire_year, salary,
       LEAD(salary, 2) OVER(
           PARTITION BY dept ORDER BY hire_year, name
       ) AS salary_two_positions_after
FROM employees;
