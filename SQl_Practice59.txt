Q16. Give a dense rank to employees based on salary.
SELECT name, salary,
DENSE_RANK() OVER(
    ORDER BY salary DESC
) AS salary_rank
FROM employees;

Q17. Give a unique row number to employees based on salary.
SELECT name, salary,
ROW_NUMBER() OVER(
    ORDER BY salary DESC
) AS row_num
FROM employees;

Q18. Show the previous employee's salary.
SELECT name, salary,
LAG(salary) OVER(
    ORDER BY hire_year
) AS previous_salary
FROM employees;

Q19. Show the next employee's salary.
SELECT name, salary,
LEAD(salary) OVER(
    ORDER BY hire_year
) AS next_salary
FROM employees;

Q20. Find the difference between current salary and previous salary.
SELECT name, salary,
salary - LAG(salary) OVER(
    ORDER BY hire_year
) AS salary_difference
FROM employees;
