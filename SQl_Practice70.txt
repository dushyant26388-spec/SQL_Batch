Q16. Divide employees into three salary groups, from highest to lowest.

SELECT name, salary,
       NTILE(3) OVER(
           ORDER BY salary DESC
       ) AS salary_group
FROM employees;

Q17. Calculate the total salary of the current employee and the previous two employees, ordered by hire year.

SELECT name, hire_year, salary,
       SUM(salary) OVER(
           ORDER BY hire_year, name
           ROWS BETWEEN 2 PRECEDING AND CURRENT ROW
       ) AS rolling_total
FROM employees;

Q18. Find the average salary of the current employee and the next two employees.

SELECT name, hire_year, salary,
       AVG(salary) OVER(
           ORDER BY hire_year, name
           ROWS BETWEEN CURRENT ROW AND 2 FOLLOWING
       ) AS next_three_avg
FROM employees;

Q19. Find the difference between each employee's salary and the lowest salary in the company.

SELECT name, salary,
       salary - MIN(salary) OVER() AS salary_difference
FROM employees;

Q20. Show each employee's rank within their department and their overall company rank.

SELECT name, dept, salary,
       RANK() OVER(
           PARTITION BY dept
           ORDER BY salary DESC
       ) AS dept_rank,
       RANK() OVER(
           ORDER BY salary DESC
       ) AS company_rank
FROM employees;
