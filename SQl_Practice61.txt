Q6. Show the number of employees working in each department.
SELECT name, dept,
       COUNT(*) OVER(PARTITION BY dept) AS employee_count
FROM employees;
Q7. Find each employee's percentage contribution to their department's total salary.
SELECT name, dept, salary,
       ROUND(
           salary * 100.0 /
           SUM(salary) OVER(PARTITION BY dept),
           2
       ) AS salary_percentage
FROM employees;
Q8. Find the difference between each employee's salary and the highest salary in their department.
SELECT name, dept, salary,
       MAX(salary) OVER(PARTITION BY dept) - salary AS difference
FROM employees;
Q9. Rank employees by salary within each department.
SELECT name, dept, salary,
       RANK() OVER(
           PARTITION BY dept
           ORDER BY salary DESC
       ) AS dept_rank
FROM employees;
Q10. Give each employee a unique number based on salary within their department.
SELECT name, dept, salary,
       ROW_NUMBER() OVER(
           PARTITION BY dept
           ORDER BY salary DESC
       ) AS row_num
FROM employees;
