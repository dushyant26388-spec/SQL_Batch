Q91. Calculate a 3-row centered salary average.
SELECT name, salary,
AVG(salary) OVER(
    ORDER BY hire_year
    ROWS BETWEEN 1 PRECEDING AND 1 FOLLOWING
) AS centered_avg
FROM employees;

Q92. Calculate the total salary from the current row and next 2 rows.
SELECT name, salary,
SUM(salary) OVER(
    ORDER BY hire_year
    ROWS BETWEEN CURRENT ROW AND 2 FOLLOWING
) AS next_3_total
FROM employees;

Q93. Calculate the average salary from the previous row, current row and next row.
SELECT name, salary,
AVG(salary) OVER(
    ORDER BY hire_year
    ROWS BETWEEN 1 PRECEDING AND 1 FOLLOWING
) AS moving_avg
FROM employees;

Q94. Show the total salary of all employees on every row using an explicit frame.
SELECT name, salary,
SUM(salary) OVER(
    ORDER BY hire_year
    ROWS BETWEEN UNBOUNDED PRECEDING
    AND UNBOUNDED FOLLOWING
) AS total_salary
FROM employees;

Q95. Find each employee's salary percentage of the company total.
SELECT name, salary,
ROUND(
    salary * 100.0 / SUM(salary) OVER(),
    2
) AS salary_percentage
FROM employees;
