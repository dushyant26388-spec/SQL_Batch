Q26. Find the second-lowest salary.
SELECT MIN(salary) AS second_lowest
FROM employees
WHERE salary > (
    SELECT MIN(salary)
    FROM employees
);

Q27. Find employees whose salary is greater than the average salary of IT employees.
SELECT emp_name, salary
FROM employees
WHERE salary > (
    SELECT AVG(e.salary)
    FROM employees e
    JOIN departments d
    ON e.dept_id = d.dept_id
    WHERE d.dept_name = 'IT'
);

Q28. Find employees whose salary is greater than the average salary of HR employees.
SELECT emp_name, salary
FROM employees
WHERE salary > (
    SELECT AVG(e.salary)
    FROM employees e
    JOIN departments d
    ON e.dept_id = d.dept_id
    WHERE d.dept_name = 'HR'
);

Q29. Find the department having the highest average salary.
SELECT d.dept_name, AVG(e.salary) AS avg_salary
FROM employees e
JOIN departments d
ON e.dept_id = d.dept_id
GROUP BY d.dept_name
ORDER BY avg_salary DESC
LIMIT 1;

Q30. Find the department having the lowest average salary.
SELECT d.dept_name, AVG(e.salary) AS avg_salary
FROM employees e
JOIN departments d
ON e.dept_id = d.dept_id
GROUP BY d.dept_name
ORDER BY avg_salary
LIMIT 1;
