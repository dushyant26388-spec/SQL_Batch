Q36. Find employees who belong to the department with the highest salary total.
SELECT name, department_id, salary
FROM Employee
WHERE department_id = (
    SELECT department_id
    FROM Employee
    GROUP BY department_id
    ORDER BY SUM(salary) DESC
    LIMIT 1
);

Q37. Find employees whose salary is greater than the average salary of the IT department.
SELECT name, salary
FROM Employee
WHERE salary > (
    SELECT AVG(e.salary)
    FROM Employee e
    JOIN Department d
    ON e.department_id = d.department_id
    WHERE d.department_name = 'IT'
);

Q38. Find employees who belong to the same department as the highest-paid employee.
SELECT name, department_id
FROM Employee
WHERE department_id = (
    SELECT department_id
    FROM Employee
    WHERE salary = (
        SELECT MAX(salary)
        FROM Employee
    )
);

Q39. Find employees whose salary is greater than every employee in the IT department.
SELECT name, salary
FROM Employee
WHERE salary > (
    SELECT MAX(e.salary)
    FROM Employee e
    JOIN Department d
    ON e.department_id = d.department_id
    WHERE d.department_name = 'IT'
);

Q40. Find employees whose salary is greater than the average salary of their department.
SELECT e.name, e.salary
FROM Employee e
WHERE e.salary > (
    SELECT AVG(e2.salary)
    FROM Employee e2
    WHERE e2.department_id = e.department_id
);
