
Q6. Find employees whose salary is higher than the previous employee's salary within their department.

SELECT name, dept, salary
FROM (
    SELECT name, dept, salary,
           LAG(salary) OVER(
               PARTITION BY dept
               ORDER BY hire_year
           ) AS previous_salary
    FROM employees
) t
WHERE salary > previous_salary;

Q7. Find the third employee hired in the company.

SELECT name, hire_year
FROM (
    SELECT name, hire_year,
           ROW_NUMBER() OVER(
               ORDER BY hire_year
           ) AS rn
    FROM employees
) t
WHERE rn = 3;

Q8. Show the next employee's hire year and the gap between their hire years.

SELECT name, hire_year,
       LEAD(hire_year) OVER(
           ORDER BY hire_year
       ) AS next_hire_year,
       LEAD(hire_year) OVER(
           ORDER BY hire_year
       ) - hire_year AS year_gap
FROM employees;

Q9. Find employees whose salary is higher than the company average.

SELECT name, salary
FROM (
    SELECT name, salary,
           AVG(salary) OVER() AS company_avg
    FROM employees
) t
WHERE salary > company_avg;

Q10. Find employees whose salary is lower than the company average.

SELECT name, salary
FROM (
    SELECT name, salary,
           AVG(salary) OVER() AS company_avg
    FROM employees
) t
WHERE salary < company_avg;
