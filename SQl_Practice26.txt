Q11. Find employees earning more than the average salary.
SELECT name, salary
FROM Employee
WHERE salary > (
    SELECT AVG(salary)
    FROM Employee
);
Q12. Find employees earning less than the average salary.
SELECT name, salary
FROM Employee
WHERE salary < (
    SELECT AVG(salary)
    FROM Employee
);
Q13. Find employees earning more than Rahul.
SELECT name, salary
FROM Employee
WHERE salary > (
    SELECT salary
    FROM Employee
    WHERE name = 'Rahul'
);
Q14. Find employees having the same salary as Rahul.
SELECT name, salary
FROM Employee
WHERE salary = (
    SELECT salary
    FROM Employee
    WHERE name = 'Rahul'
);
Q15. Find employees working in the same department as Rahul.
SELECT name
FROM Employee
WHERE department_id = (
    SELECT department_id
    FROM Employee
    WHERE name = 'Rahul'
);
