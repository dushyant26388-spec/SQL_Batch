Q36. Find employees whose salary is higher than the average salary of the company and display their department.
SELECT e.emp_name, e.salary, d.dept_name
FROM employees e
JOIN departments d
ON e.dept_id = d.dept_id
WHERE e.salary > (
    SELECT AVG(salary)
    FROM employees
);

Q37. Find departments that have no employees.
SELECT d.dept_name
FROM departments d
LEFT JOIN employees e
ON d.dept_id = e.dept_id
WHERE e.emp_id IS NULL;


Q38. Find employees whose department has the highest total salary.
SELECT e.emp_name, e.salary, d.dept_name
FROM employees e
JOIN departments d
ON e.dept_id = d.dept_id
WHERE e.dept_id = (
    SELECT dept_id
    FROM employees
    GROUP BY dept_id
    ORDER BY SUM(salary) DESC
    LIMIT 1
);
Q39. Find employees working in the department with the highest average salary.
SELECT e.emp_name, d.dept_name, e.salary
FROM employees e
JOIN departments d
ON e.dept_id = d.dept_id
WHERE e.dept_id = (
    SELECT dept_id
    FROM employees
    GROUP BY dept_id
    ORDER BY AVG(salary) DESC
    LIMIT 1
);
Q40. Find employees who have the same salary as another employee.
SELECT e1.emp_name, e1.salary
FROM employees e1
JOIN employees e2
ON e1.salary = e2.salary
AND e1.emp_id <> e2.emp_id;
