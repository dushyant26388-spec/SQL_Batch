Q6. Find employees who joined first in their department.
SELECT name, dept, hire_year
FROM (
    SELECT name, dept, hire_year,
    ROW_NUMBER() OVER(
        PARTITION BY dept
        ORDER BY hire_year ASC
    ) AS rn
    FROM employees
) t
WHERE rn = 1;

Q7. Show RANK, DENSE_RANK and ROW_NUMBER together for each employee within their department.
SELECT name, dept, salary,
RANK() OVER(
    PARTITION BY dept
    ORDER BY salary DESC
) AS rnk,
DENSE_RANK() OVER(
    PARTITION BY dept
    ORDER BY salary DESC
) AS dense_rnk,
ROW_NUMBER() OVER(
    PARTITION BY dept
    ORDER BY salary DESC
) AS row_num
FROM employees
ORDER BY dept, salary DESC;

Q8. Find the 2nd highest paid employee in each department using DENSE_RANK.
SELECT name, dept, salary
FROM (
    SELECT name, dept, salary,
    DENSE_RANK() OVER(
        PARTITION BY dept
        ORDER BY salary DESC
    ) AS dr
    FROM employees
) t
WHERE dr = 2;

Q9. For each Alice sale, show the previous month's sales amount.
SELECT emp_name, sale_month, amount,
LAG(amount, 1, 0) OVER(
    ORDER BY sale_id
) AS prev_amount
FROM sales
WHERE emp_name = 'Alice';

Q10. Show each sale along with the next month's sales.
SELECT emp_name, sale_month, amount,
LEAD(amount, 1, 0) OVER(
    PARTITION BY emp_name
    ORDER BY sale_id
) AS next_month
FROM sales;
