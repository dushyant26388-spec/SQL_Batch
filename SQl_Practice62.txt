Q11. Show the previous employee's salary based on hire year.
SELECT name, hire_year, salary,
       LAG(salary) OVER(
           ORDER BY hire_year
       ) AS previous_salary
FROM employees;
Q12. Show the next employee's salary based on hire year.
SELECT name, hire_year, salary,
       LEAD(salary) OVER(
           ORDER BY hire_year
       ) AS next_salary
FROM employees;
Q13. Find the salary change from the previous employee.
SELECT name, hire_year, salary,
       salary - LAG(salary) OVER(
           ORDER BY hire_year
       ) AS salary_change
FROM employees;
Q14. Find the running total of salaries according to hire year.
SELECT name, hire_year, salary,
       SUM(salary) OVER(
           ORDER BY hire_year
       ) AS running_total
FROM employees;
Q15. Find the running average salary according to hire year.
SELECT name, hire_year, salary,
       AVG(salary) OVER(
           ORDER BY hire_year
       ) AS running_average
FROM employees;
