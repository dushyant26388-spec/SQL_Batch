Q46. Calculate the maximum sales value from the current day and previous day.
SELECT day_num, sale_amount,
MAX(sale_amount) OVER(
    ORDER BY day_num
    ROWS BETWEEN 1 PRECEDING AND CURRENT ROW
) AS max_last_2_days
FROM daily_sales;

Q47. Calculate the total sales from the current day and next 2 days.
SELECT day_num, sale_amount,
SUM(sale_amount) OVER(
    ORDER BY day_num
    ROWS BETWEEN CURRENT ROW AND 2 FOLLOWING
) AS next_3_days_total
FROM daily_sales;

Q48. Calculate the average sales for the current day and next day.
SELECT day_num, sale_amount,
AVG(sale_amount) OVER(
    ORDER BY day_num
    ROWS BETWEEN CURRENT ROW AND 1 FOLLOWING
) AS avg_this_and_next
FROM daily_sales;

Q49. Calculate the grand total sales on every row.
SELECT day_num, sale_amount,
SUM(sale_amount) OVER(
    ORDER BY day_num
    ROWS BETWEEN UNBOUNDED PRECEDING
    AND UNBOUNDED FOLLOWING
) AS grand_total
FROM daily_sales;

Q50. Calculate the maximum sales value across all days.
SELECT day_num, sale_amount,
MAX(sale_amount) OVER(
    ORDER BY day_num
    ROWS BETWEEN UNBOUNDED PRECEDING
    AND UNBOUNDED FOLLOWING
) AS overall_max
FROM daily_sales;
