Q1. Show each employee's salary along with the total salary of their department.
SELECT name, dept, salary,
       SUM(salary) OVER(PARTITION BY dept) AS dept_total
FROM employees;
Q2. Show each employee's salary along with the average salary of their department.
SELECT name, dept, salary,
       AVG(salary) OVER(PARTITION BY dept) AS dept_avg
FROM employees;
Q3. Find how much higher or lower each employee's salary is compared with their department average.
SELECT name, dept, salary,
       salary - AVG(salary) OVER(PARTITION BY dept) AS difference
FROM employees;
Q4. Show the highest salary in each department beside every employee.
SELECT name, dept, salary,
       MAX(salary) OVER(PARTITION BY dept) AS highest_salary
FROM employees;
Q5. Show the lowest salary in each department beside every employee.
SELECT name, dept, salary,
       MIN(salary) OVER(PARTITION BY dept) AS lowest_salary
FROM employees;
